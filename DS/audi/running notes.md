Underflow happens when u pop elements even after all the elements have been popped, i.e repeated popping of elements even after having no elements leads to stack underflow..............***fixed capacity***class stack{&nbsp;	int a\[];&nbsp;	int top;&nbsp;	int capacity;&nbsp;	stack(int b){&nbsp;		capacity = b;&nbsp;	 	a= new int\[capacity];&nbsp;	}top = -1;&nbsp;	bool isfull(){&nbsp;		if(top!=capacity){&nbsp;			return false;&nbsp;		}&nbsp;		return true;&nbsp;	}&nbsp;	void push(int x){&nbsp;		top++;&nbsp;		a\[top]=x;&nbsp;	}&nbsp;	void pop(){&nbsp;		top--; 	}};.........................***no fixed capacity***class stack1{&nbsp;	vector<int> a;&nbsp;	int top;};.........................
